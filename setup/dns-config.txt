------------------------------------------------------------
  2. BREVO — Authentification emails (SPF / DKIM / DMARC)
------------------------------------------------------------

# SPF — Autoriser Brevo à envoyer pour creatis.app
Type : TXT
Sous-domaine : @ (racine)
Valeur : "v=spf1 include:spf.brevo.com mx ~all"
TTL : 3600
Note : Si tu as déjà un SPF pour OVH/Gmail, COMBINER :
       "v=spf1 include:spf.brevo.com include:mx.ovh.com ~all"

# DKIM Brevo — À récupérer dans Brevo Dashboard
# Brevo → Settings → Senders & IPs → Domains → Authenticate a Domain
# Brevo génère une valeur unique à copier ci-dessous :
Type : TXT
Sous-domaine : mail._domainkey
Valeur : [VALEUR FOURNIE PAR BREVO — format "v=DKIM1; k=rsa; p=..."]
TTL : 3600

# Brevo DKIM link (alternative selon config Brevo)
Type : CNAME
Sous-domaine : mail._domainkey
Valeur : [ex : mail._domainkey.brevo.com.]
TTL : 3600

# DMARC — Politique anti-usurpation
Type : TXT
Sous-domaine : _dmarc
Valeur : "v=DMARC1; p=quarantine; rua=mailto:dmarc@creatis.app; pct=100"
TTL : 3600
Note : Commencer avec p=none pour monitoring, puis passer à quarantine

# Vérification SPF :
#   dig TXT creatis.app | grep spf
# Vérification DKIM :
#   dig TXT mail._domainkey.creatis.app
# Testeur complet : https://www.mail-tester.com
